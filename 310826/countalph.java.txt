public class Main
{
	public static void main(String[] args) {
	    String s="apple";
	    int[] arr=new int[26];
	    /*int n='a';

	    for(int i=0;i<s.length();i++){
	        int index=s.charAt(i)-'a';

	        }
	    }
	}
}
