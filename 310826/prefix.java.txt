public class Main
{
	public static void main(String[] args) {
	    String s="*+123";
	    int n=s.length();
	    int out=s.charAt((n/2))-'0';
	    for(int i=0;i<n/2;i++){
	        switch(s.charAt(i)){
	            case '-':
	                out-=((s.charAt(i+(n/2)+1))-'0');
	                break;
	            case '+':
	                out+=(s.charAt(i+(n/2)+1))-'0';
	                break;
	            case '*':
	                out*=((s.charAt(i+(n/2)+1))-'0');
	                break;
	            case '/':
	                out/=(s.charAt(i+(n/2)+1))-'0';
	                break;
	           default:
	                System.out.println("invalid");
	        }
	    }
		System.out.println(out);
	}
}
