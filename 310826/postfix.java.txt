public class Main
{
	public static void main(String[] args) {
	    String s="123*+";
	    int n=s.length();
	    int out=s.charAt(0)-'0';
	    for(int i=(n/2)+1;i<n;i++){
	        switch(s.charAt(i)){
	            case '-':
	                out-=(s.charAt(i-(n/2)))-'0';
	                break;
	            case '+':
	                out+=(s.charAt(i-(n/2)))-'0';
	                break;
	            case '*':
	                out*=(s.charAt(i-(n/2)))-'0';
	                break;
	            case '/':
	                out/=(s.charAt(i-(n/2)))-'0';
	                break;
	           default:
	                System.out.println("invalid");
	        }
	    }
		System.out.println(out);
	}
}
